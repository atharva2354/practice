import pandas as pd
import numpy as np
import tensorflow as tf

# Load dataset
df = pd.read_csv('network.csv')

# Preprocessing
X = df.select_dtypes(include='number').fillna(0).values

# Select normal data
normal = X[df['label'] == 'normal']

# Build autoencoder
model = tf.keras.Sequential([
    tf.keras.layers.Dense(16, activation='relu', input_shape=(X.shape[1],)),
    tf.keras.layers.Dense(8, activation='relu'),
    tf.keras.layers.Dense(16, activation='relu'),
    tf.keras.layers.Dense(X.shape[1])
])

# Compile
model.compile(optimizer='adam', loss='mse')

# Train
model.fit(normal, normal, epochs=5)

# Evaluate
model.evaluate(normal, normal)

# Calculate threshold
pred = model.predict(normal)
error = np.mean((normal - pred)**2, axis=1)
threshold = np.percentile(error, 95)

# Detect attacks
pred = model.predict(X)
error = np.mean((X - pred)**2, axis=1)

anomalies = error > threshold

print("Threshold:", threshold)
print("Attacks detected:", np.sum(anomalies))
